template <typename T>
class SmartPtr {
    T* ptr;
public:
    explicit SmartPtr(T* p = nullptr) : ptr(p) {}
    ~SmartPtr() { delete ptr; }
    // Rule of Five: RAII Move Semantics
    SmartPtr(SmartPtr&& o) noexcept : ptr(o.release()) {}
    T& operator*() const { return *ptr; }
};
